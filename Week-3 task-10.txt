import java.io.*;
import java.util.*;
import java.util.stream.*;
import static java.util.stream.Collectors.toList;

class Result {

    /*
     * Complete the 'matrixRotation' function below.
     *
     * The function accepts following parameters:
     *  1. 2D_INTEGER_ARRAY matrix
     *  2. INTEGER r
     */

    public static void matrixRotation(List<List<Integer>> matrix, int r) {

        int m = matrix.size();
        int n = matrix.get(0).size();

        int layers = Math.min(m, n) / 2;

        for (int layer = 0; layer < layers; layer++) {

            List<Integer> values = new ArrayList<>();

            // Top row
            for (int j = layer; j < n - layer; j++) {
                values.add(matrix.get(layer).get(j));
            }

            // Right column
            for (int i = layer + 1; i < m - layer; i++) {
                values.add(matrix.get(i).get(n - layer - 1));
            }

            // Bottom row
            for (int j = n - layer - 2; j >= layer; j--) {
                values.add(matrix.get(m - layer - 1).get(j));
            }

            // Left column
            for (int i = m - layer - 2; i > layer; i--) {
                values.add(matrix.get(i).get(layer));
            }

            // Calculate effective rotation
            int rotate = r % values.size();

            // Rotate counter-clockwise
            List<Integer> rotated = new ArrayList<>();

            for (int i = rotate; i < values.size(); i++) {
                rotated.add(values.get(i));
            }

            for (int i = 0; i < rotate; i++) {
                rotated.add(values.get(i));
            }

            int index = 0;

            // Put back - Top row
            for (int j = layer; j < n - layer; j++) {
                matrix.get(layer).set(j, rotated.get(index++));
            }

            // Put back - Right column
            for (int i = layer + 1; i < m - layer; i++) {
                matrix.get(i).set(n - layer - 1, rotated.get(index++));
            }

            // Put back - Bottom row
            for (int j = n - layer - 2; j >= layer; j--) {
                matrix.get(m - layer - 1).set(j, rotated.get(index++));
            }

            // Put back - Left column
            for (int i = m - layer - 2; i > layer; i--) {
                matrix.get(i).set(layer, rotated.get(index++));
            }
        }

        // Print the rotated matrix
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {

                System.out.print(matrix.get(i).get(j));

                if (j < n - 1) {
                    System.out.print(" ");
                }
            }

            System.out.println();
        }
    }
}

public class Solution {

    public static void main(String[] args) throws IOException {

        BufferedReader bufferedReader =
            new BufferedReader(new InputStreamReader(System.in));

        String[] firstMultipleInput =
            bufferedReader.readLine()
                .replaceAll("\\s+$", "")
                .split(" ");

        int m = Integer.parseInt(firstMultipleInput[0]);
        int n = Integer.parseInt(firstMultipleInput[1]);
        int r = Integer.parseInt(firstMultipleInput[2]);

        List<List<Integer>> matrix = new ArrayList<>();

        IntStream.range(0, m).forEach(i -> {
            try {
                matrix.add(
                    Stream.of(
                        bufferedReader.readLine()
                            .replaceAll("\\s+$", "")
                            .split(" ")
                    )
                    .map(Integer::parseInt)
                    .collect(toList())
                );
            } catch (IOException ex) {
                throw new RuntimeException(ex);
            }
        });

        Result.matrixRotation(matrix, r);

        bufferedReader.close();
    }
}
